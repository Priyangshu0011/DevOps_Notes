Answers:

(For Task)
docker exec -it apache-web bash
mkdir /project
cd /project
touch report.txt
echo "This is our container interaction with our host machine." > report.txt
cat report.txt
exit
docker cp apache-web:/project/report.txt ~/Desktop/report.txt
docker exec -it apache-web bash
cd /project
touch notes.txt
echo "Additional notes inside container." >> notes.txt
ls /project
exit

(For Practice Questions)
docker run -dit -e ENV_MODE=production nginx
docker logs -f my_app
docker start web_server
docker stop web_server
